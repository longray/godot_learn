# 第 7 课：敌人追击与简单状态机

本课目标：

1. 给敌人添加状态机。
2. 敌人会巡逻。
3. 玩家进入检测范围后，敌人发现玩家。
4. 敌人追击玩家。
5. 玩家跑远或躲到墙后，敌人会短暂搜索。
6. 搜索失败后，敌人返回巡逻点。
7. 敌人不会隔着墙直接发现玩家。

---

# 1. 本课敌人行为设计

我们会使用一个非常简单的状态机：

```text
PATROL  巡逻
CHASE   追击
RETURN  返回巡逻点
```

状态转换：

```text
PATROL：
    如果看到玩家 -> CHASE

CHASE：
    如果玩家太远或丢失视野一段时间 -> RETURN

RETURN：
    如果再次看到玩家 -> CHASE
    如果回到巡逻点 -> PATROL
```

敌人是否能看到玩家，不只判断距离，还会做简单的“视线检测”：

```text
如果玩家距离很近，但中间隔着墙，
敌人不会立刻发现玩家。
```

```gdscript
extends CharacterBody2D

# 第 7 课：会追击玩家的巡逻敌人

enum EnemyState {
    PATROL,
    CHASE,
    RETURN
}

# ---------- 基础移动 ----------

@export var speed: float = 70.0
@export var wait_time: float = 0.8
@export var arrival_distance: float = 4.0
@export var collision_radius: float = 4.0

# ---------- 接触伤害 ----------

@export var contact_damage: int = 1

# ---------- 生命 ----------

@export var max_health: int = 2

# ---------- 追击参数 ----------

@export var detection_range: float = 90.0
@export var lose_range: float = 150.0
@export var lose_sight_time: float = 0.8
@export var chase_speed_multiplier: float = 1.4

# ---------- 外部数据 ----------

var dungeon: Node
var patrol_points: Array = []

# ---------- 巡逻状态 ----------

var current_point_index: int = 0
var is_waiting: bool = false
var wait_remaining: float = 0.0

# ---------- 生命状态 ----------

var health: int = 2
var dead: bool = false

# ---------- AI 状态 ----------

var state: EnemyState = EnemyState.PATROL

var player: Node2D = null
var last_known_player_position := Vector2.ZERO
var time_since_seen: float = 0.0
var stuck_time: float = 0.0

var home_position := Vector2.ZERO

@onready var hitbox: Area2D = $Hitbox


func _ready() -> void:
    health = max_health

    if hitbox:
        hitbox.body_entered.connect(_on_hitbox_body_entered)


func setup(dungeon_reference: Node, points: Array) -> void:
    dungeon = dungeon_reference
    patrol_points = points

    health = max_health
    dead = false

    state = EnemyState.PATROL

    is_waiting = false
    wait_remaining = 0.0
    stuck_time = 0.0
    time_since_seen = 0.0

    if patrol_points.is_empty():
        home_position = global_position
        last_known_player_position = global_position
        return

    global_position = patrol_points[0]
    home_position = patrol_points[0]
    last_known_player_position = global_position

    current_point_index = 0

    if patrol_points.size() > 1:
        current_point_index = 1


func take_damage(amount: int, source_position: Vector2 = Vector2.ZERO) -> void:
    if dead:
        return

    health -= amount

    print("敌人受击，剩余生命：", health)

    if health <= 0:
        dead = true
        _die()
        return

    _flash()


func _die() -> void:
    if dungeon and dungeon.has_method("on_enemy_died"):
        dungeon.on_enemy_died(self, global_position)

    queue_free()


func _flash() -> void:
    modulate = Color(1.0, 0.4, 0.4)

    await get_tree().create_timer(0.08).timeout

    if is_inside_tree() and not dead:
        modulate = Color.WHITE


func _physics_process(delta: float) -> void:
    if dead:
        velocity = Vector2.ZERO
        move_and_slide()
        return

    if dungeon == null or not dungeon.has_method("is_world_position_walkable"):
        velocity = Vector2.ZERO
        move_and_slide()
        return

    var current_player := _get_player()

    _update_state(delta, current_player)

    match state:
        EnemyState.PATROL:
            _process_patrol(delta)
        EnemyState.CHASE:
            _process_chase(delta)
        EnemyState.RETURN:
            _process_return(delta)


# =========================
# 玩家获取
# =========================

func _get_player() -> Node2D:
    if player == null or not is_instance_valid(player):
        player = get_tree().get_first_node_in_group("player") as Node2D

    return player


# =========================
# 状态转换
# =========================

func _update_state(delta: float, current_player: Node2D) -> void:
    match state:
        EnemyState.PATROL:
            if _is_player_visible(current_player, detection_range):
                _start_chase(current_player)

        EnemyState.CHASE:
            if current_player == null or not is_instance_valid(current_player):
                state = EnemyState.RETURN
                stuck_time = 0.0
                return

            if _is_player_visible(current_player, lose_range):
                last_known_player_position = current_player.global_position
                time_since_seen = 0.0
            else:
                time_since_seen += delta

            var distance_to_player := global_position.distance_to(current_player.global_position)

            if distance_to_player > lose_range or time_since_seen > lose_sight_time:
                state = EnemyState.RETURN
                stuck_time = 0.0

        EnemyState.RETURN:
            if _is_player_visible(current_player, detection_range):
                _start_chase(current_player)


func _start_chase(target_player: Node2D) -> void:
    state = EnemyState.CHASE
    is_waiting = false
    stuck_time = 0.0
    time_since_seen = 0.0
    last_known_player_position = target_player.global_position


# =========================
# 巡逻
# =========================

func _process_patrol(delta: float) -> void:
    if patrol_points.is_empty():
        velocity = Vector2.ZERO
        move_and_slide()
        return

    if patrol_points.size() == 1:
        velocity = Vector2.ZERO
        move_and_slide()
        return

    if is_waiting:
        wait_remaining -= delta
        velocity = Vector2.ZERO

        if wait_remaining <= 0.0:
            is_waiting = false
            current_point_index = (current_point_index + 1) % patrol_points.size()

        move_and_slide()
        return

    var target: Vector2 = patrol_points[current_point_index]
    var arrived := _move_towards(target, speed, delta)

    if arrived:
        is_waiting = true
        wait_remaining = wait_time
        velocity = Vector2.ZERO
    elif velocity == Vector2.ZERO:
        # 如果巡逻被墙卡住，就切换下一个巡逻点
        current_point_index = (current_point_index + 1) % patrol_points.size()

    move_and_slide()


# =========================
# 追击
# =========================

func _process_chase(delta: float) -> void:
    var target := last_known_player_position
    var chase_speed := speed * chase_speed_multiplier

    var arrived := _move_towards(target, chase_speed, delta)

    if arrived:
        # 如果到达最后已知位置，但已经看不到玩家，就准备返回
        if time_since_seen > 0.05:
            state = EnemyState.RETURN
            stuck_time = 0.0
    elif velocity == Vector2.ZERO:
        stuck_time += delta

        if stuck_time > 0.5:
            state = EnemyState.RETURN
            stuck_time = 0.0
    else:
        stuck_time = 0.0

    move_and_slide()


# =========================
# 返回
# =========================

func _process_return(delta: float) -> void:
    var target := home_position
    var target_index := 0

    if not patrol_points.is_empty():
        target_index = _get_closest_patrol_point_index()
        target = patrol_points[target_index]

    var arrived := _move_towards(target, speed, delta)

    if arrived:
        state = EnemyState.PATROL
        current_point_index = target_index
        is_waiting = false
        stuck_time = 0.0
    elif velocity == Vector2.ZERO:
        stuck_time += delta

        if stuck_time > 0.8:
            # 如果返回路径一直被卡住，就强制回到巡逻状态
            state = EnemyState.PATROL
            stuck_time = 0.0
    else:
        stuck_time = 0.0

    move_and_slide()


func _get_closest_patrol_point_index() -> int:
    if patrol_points.is_empty():
        return 0

    var best_index := 0
    var best_distance := 999999999.0

    for i in patrol_points.size():
        var distance := global_position.distance_squared_to(patrol_points[i])

        if distance < best_distance:
            best_distance = distance
            best_index = i

    return best_index


# =========================
# 移动辅助
# =========================

func _move_towards(target: Vector2, move_speed: float, delta: float) -> bool:
    var direction := target - global_position
    var distance := direction.length()

    if distance <= arrival_distance:
        velocity = Vector2.ZERO
        return true

    var desired_velocity := direction.normalized() * move_speed
    var next_position := global_position + desired_velocity * delta

    if dungeon.is_world_position_walkable(next_position, collision_radius):
        velocity = desired_velocity
        return false

    # 简单贴墙滑动
    var x_only_position := Vector2(next_position.x, global_position.y)
    var y_only_position := Vector2(global_position.x, next_position.y)

    if dungeon.is_world_position_walkable(x_only_position, collision_radius):
        velocity = Vector2(desired_velocity.x, 0.0)
        return false

    if dungeon.is_world_position_walkable(y_only_position, collision_radius):
        velocity = Vector2(0.0, desired_velocity.y)
        return false

    velocity = Vector2.ZERO
    return false


# =========================
# 视线检测
# =========================

func _is_player_visible(target: Node2D, range_limit: float) -> bool:
    if target == null or not is_instance_valid(target):
        return false

    if dungeon == null or not dungeon.has_method("is_world_position_walkable"):
        return false

    var distance := global_position.distance_to(target.global_position)

    if distance > range_limit:
        return false

    # 沿敌人与玩家之间采样，检查是否被墙挡住
    var steps := int(distance / 8.0) + 1

    for i in range(steps + 1):
        var t := float(i) / float(steps)
        var sample := global_position.lerp(target.global_position, t)

        if not dungeon.is_world_position_walkable(sample, 0.0):
            return false

    return true


# =========================
# 接触伤害
# =========================

func _on_hitbox_body_entered(body: Node2D) -> void:
    if dead:
        return

    if body.is_in_group("player") and body.has_method("take_damage"):
        body.take_damage(contact_damage, global_position)
```

# 5. 本课核心知识讲解

---

## 5.1 什么是状态机？

状态机就是：

```text
当前状态
+ 状态规则
+ 状态行为
```

本课敌人的状态：

```text
PATROL
CHASE
RETURN
```

每个状态有自己的行为：

```text
PATROL：巡逻
CHASE：追玩家
RETURN：回家
```

每个状态也有转换规则：

```text
PATROL 看到玩家 -> CHASE
CHASE 丢失玩家 -> RETURN
RETURN 看到玩家 -> CHASE
RETURN 到家 -> PATROL
```

这是游戏 AI 最基础也最实用的结构。

---

## 5.2 为什么敌人需要“最后已知位置”？

如果敌人只看得到当前玩家位置，那么玩家一躲到墙后，敌人就会立刻失去目标。

这样会显得很笨。

所以我们加入：

```gdscript
last_known_player_position
```

敌人丢失视野后，还会先去最后看到玩家的位置找一下。

这会让敌人显得更聪明。

---

## 5.3 为什么要做视线检测？

如果不做视线检测：

```text
只要玩家距离近，
即使隔着墙，敌人也会发现玩家。
```

这会非常不自然。

本课使用简单采样：

```gdscript
global_position.lerp(target.global_position, t)
```

在敌人与玩家之间取多个点，然后检查这些点是否是地板。

如果中间有墙：

```gdscript
dungeon.is_world_position_walkable(sample, 0.0)
```

会返回 `false`，敌人就看不到玩家。

---

## 5.4 为什么敌人会卡住？

程序化地图里经常出现：

- 窄角。
- 不规则走廊。
- 房间边缘。
- 装饰物碰撞。
- 寻路目标和实际可走路径不一致。

本课敌人使用了简单策略：

```text
如果移动时速度一直为 0，
说明可能被卡住。
超过一段时间后，切换状态。
```

这虽然不是完美寻路，但可以让原型稳定运行。

后续可以升级为：

- `AStarGrid2D` 路径。
- `NavigationAgent2D`。
- 动态避障。
- 路径重计算。
- 卡住时瞬移回巡逻点。

# 6. 参数调节建议

你可以在 `Enemy` 的 Inspector 里调节这些参数。

---

## 6.1 Detection Range

```text
Detection Range = 90
```

敌人发现玩家的距离。

调大：

```text
敌人更警觉
```

调小：

```text
敌人更迟钝
```

---

## 6.2 Lose Range

```text
Lose Range = 150
```

敌人放弃追击的距离。

建议：

```text
Lose Range > Detection Range
```

否则敌人刚发现玩家，玩家稍微拉开距离，敌人就放弃了。

---

## 6.3 Lose Sight Time

```text
Lose Sight Time = 0.8
```

敌人丢失视野后继续搜索的时间。

调大：

```text
敌人更执着
```

调小：

```text
敌人更容易放弃
```

---

## 6.4 Chase Speed Multiplier

```text
Chase Speed Multiplier = 1.4
```

追击速度倍率。

例如：

```text
speed = 70
chase_speed_multiplier = 1.4
```

追击速度：

```text
98
```

如果玩家速度是：

```text
140
```

玩家仍然可以逃跑。

如果你想让敌人更有压迫感，可以调到：

```text
1.7
```

或者：

```text
1.9
```

# 7. 常见问题排查

---

## 问题 1：敌人不会追击玩家

检查：

1. 玩家是否在 `player` 组。
2. 敌人是否已经通过 `Main` 设置 `dungeon`。
3. `Detection Range` 是否太小。
4. 玩家和敌人之间是否隔墙。
5. 控制台是否有脚本错误。

你可以在 `_start_chase()` 里加一句打印：

```gdscript
print("敌人发现玩家！")
```

方便调试。

---

## 问题 2：敌人隔墙发现玩家

检查：

```gdscript
_is_player_visible()
```

是否被正常调用。

如果地图格子太大，采样间隔 `8.0` 可能太粗。

可以改成：

```gdscript
var steps := int(distance / 4.0) + 1
```

这样视线检测更细，但性能消耗稍高。

---

## 问题 3：敌人追击时穿墙

本课敌人移动依赖：

```gdscript
dungeon.is_world_position_walkable()
```

如果敌人穿墙，检查：

1. `collision_radius` 是否太小。
2. `cell_size` 是否和地图瓦片一致。
3. 敌人速度是否过高。
4. 地图墙角是否留有缝隙。

---

## 问题 4：敌人一直在原地抖动

可能是：

- 目标点不可达。
- `collision_radius` 太大。
- 巡逻点太靠近墙。
- 敌人出生点卡在墙边。

可以尝试：

```text
Collision Radius = 3
Arrival Distance = 5
```

---

## 问题 5：敌人追到玩家后不攻击

敌人攻击不是本课新内容，它来自第 5 课的：

```text
Hitbox
```

检查：

1. `Enemy` 场景里是否有 `Hitbox`。
2. `Hitbox` 是否是 `Area2D`。
3. `Hitbox` 是否有 `CollisionShape2D`。
4. `Hitbox` 的 mask 是否能检测到玩家。
5. 玩家是否有 `take_damage()`。

推荐设置：

```text
Enemy Hitbox Collision Mask = 1
Player Collision Layer = 1
```

---

## 问题 6：敌人状态切换太频繁

例如：

```text
刚追击
又返回
又追击
```

可以调整：

```text
Lose Sight Time = 1.2
Detection Range = 80
Lose Range = 170
```

也可以后续加入“警报冷却时间”。

> **HUD 与游戏状态面板。**

我们会把当前游戏信息显示出来：

```text
生命值
金币数
钥匙状态
当前层数
宝箱数
```

这样你的项目会从“程序原型”变成更像完整游戏的形式。
