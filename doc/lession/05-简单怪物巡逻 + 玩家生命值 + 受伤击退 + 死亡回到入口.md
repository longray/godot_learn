# 3. Enemy 脚本

给 `Enemy` 根节点添加脚本：

```text
enemy.gd
```

完整代码：

```gdscript
extends CharacterBody2D

# 第 5 课：简单巡逻敌人

@export var speed: float = 70.0
@export var wait_time: float = 0.8
@export var arrival_distance: float = 4.0
@export var collision_radius: float = 4.0
@export var contact_damage: int = 1

var dungeon: Node
var patrol_points: Array = []

var current_point_index: int = 0
var is_waiting: bool = false
var wait_remaining: float = 0.0

@onready var hitbox: Area2D = $Hitbox


func _ready() -> void:
    if hitbox:
        hitbox.body_entered.connect(_on_hitbox_body_entered)


# Main 会调用这个函数
func setup(dungeon_reference: Node, points: Array) -> void:
    dungeon = dungeon_reference
    patrol_points = points

    if patrol_points.is_empty():
        return

    global_position = patrol_points[0]
    current_point_index = 0

    if patrol_points.size() > 1:
        current_point_index = 1

    is_waiting = false
    wait_remaining = 0.0


func _physics_process(delta: float) -> void:
    if dungeon == null or not dungeon.has_method("is_world_position_walkable"):
        velocity = Vector2.ZERO
        move_and_slide()
        return

    if patrol_points.is_empty():
        velocity = Vector2.ZERO
        move_and_slide()
        return

    # 只有一个巡逻点时，敌人原地待机
    if patrol_points.size() == 1:
        velocity = Vector2.ZERO
        move_and_slide()
        return

    if is_waiting:
        wait_remaining -= delta
        velocity = Vector2.ZERO

        if wait_remaining <= 0.0:
            is_waiting = false
            current_point_index = (current_point_index + 1) % patrol_points.size()

        move_and_slide()
        return

    var target: Vector2 = patrol_points[current_point_index]
    var direction: Vector2 = target - global_position
    var distance: float = direction.length()

    if distance <= arrival_distance:
        is_waiting = true
        wait_remaining = wait_time
        velocity = Vector2.ZERO
        move_and_slide()
        return

    var desired_velocity := direction.normalized() * speed
    var next_position := global_position + desired_velocity * delta

    if dungeon.is_world_position_walkable(next_position, collision_radius):
        velocity = desired_velocity
    else:
        # 如果前方不可走，就切换巡逻点
        velocity = Vector2.ZERO
        current_point_index = (current_point_index + 1) % patrol_points.size()

    move_and_slide()


func _on_hitbox_body_entered(body: Node2D) -> void:
    if body.is_in_group("player") and body.has_method("take_damage"):
        body.take_damage(contact_damage, global_position)
```

# 4. 更新 Player 脚本

打开第 3 课的：

```text

```

我们要把它升级成有生命值、受伤、击退、死亡重生的版本。

完整替换为：

```gdscript
extends CharacterBody2D

# 第 5 课：可受伤的玩家

@export var speed: float = 140.0
@export var collision_radius: float = 5.0

@export var max_health: int = 3
@export var invincibility_time: float = 0.8

var dungeon: Node

var health: int = 3
var invincible: bool = false
var knockback := Vector2.ZERO


func _ready() -> void:
    health = max_health


func reset_for_new_layer() -> void:
    health = max_health
    invincible = false
    knockback = Vector2.ZERO
    modulate.a = 1.0


func _physics_process(delta: float) -> void:
    if dungeon == null or not dungeon.has_method("is_world_position_walkable"):
        velocity = Vector2.ZERO
        move_and_slide()
        return

    var input_vector := Input.get_vector(
        "ui_left",
        "ui_right",
        "ui_up",
        "ui_down"
    )

    var desired_velocity := Vector2.ZERO

    if input_vector != Vector2.ZERO:
        desired_velocity = input_vector.normalized() * speed

    var move_velocity := desired_velocity + knockback

    # 击退随时间衰减
    knockback = knockback.move_toward(Vector2.ZERO, 500.0 * delta)

    if move_velocity == Vector2.ZERO:
        velocity = Vector2.ZERO
        move_and_slide()
        return

    var next_position := global_position + move_velocity * delta

    if dungeon.is_world_position_walkable(next_position, collision_radius):
        velocity = move_velocity
    else:
        var x_only_position := Vector2(next_position.x, global_position.y)
        var y_only_position := Vector2(global_position.x, next_position.y)

        if dungeon.is_world_position_walkable(x_only_position, collision_radius):
            velocity = Vector2(move_velocity.x, 0.0)
        elif dungeon.is_world_position_walkable(y_only_position, collision_radius):
            velocity = Vector2(0.0, move_velocity.y)
        else:
            velocity = Vector2.ZERO

    move_and_slide()


func take_damage(amount: int, source_position: Vector2) -> void:
    if invincible:
        return

    health -= amount

    print("玩家受伤，剩余生命：", health)

    if health <= 0:
        die()
        return

    _apply_knockback(source_position)
    _start_invincibility(invincibility_time)


func die() -> void:
    print("玩家死亡，回到入口。")

    health = max_health
    knockback = Vector2.ZERO

    if dungeon and dungeon.has_method("respawn_player"):
        dungeon.respawn_player()

    _start_invincibility(invincibility_time * 1.5)


func _apply_knockback(source_position: Vector2) -> void:
    var direction := global_position - source_position

    if direction == Vector2.ZERO:
        direction = Vector2.RIGHT

    knockback = direction.normalized() * 180.0


func _start_invincibility(duration: float) -> void:
    invincible = true
    modulate.a = 0.45

    await get_tree().create_timer(duration).timeout

    if is_inside_tree():
        invincible = false
        modulate.a = 1.0
```

## 5.5 添加巡逻路径生成函数

添加下面这些函数。

它们会根据地图数据给敌人生成简单巡逻路径。

```gdscript
func _make_patrol_points(cell: Vector2i) -> Array:
    var points: Array = []

    # 优先在敌人所在房间里巡逻：
    # 敌人出生点 <-> 房间中心
    var room := _find_room_containing_cell(cell)

    if room.size != Vector2i.ZERO:
        var center_cell := _room_center(room)

        if center_cell != cell and is_cell_walkable(center_cell):
            points.append(get_cell_world_position(cell))
            points.append(get_cell_world_position(center_cell))
            return points

    # 如果敌人不在房间里，或者房间中心不可用，
    # 就尝试沿上下左右找一个可走方向做短距离巡逻。
    var directions := [
        Vector2i(1, 0),
        Vector2i(-1, 0),
        Vector2i(0, 1),
        Vector2i(0, -1)
    ]

    for dir in directions:
        var forward := _scan_walkable_cells(cell, dir, 5)

        if not forward.is_empty():
            points.append(get_cell_world_position(cell))
            points.append(get_cell_world_position(forward[-1]))
            return points

        var backward := _scan_walkable_cells(cell, -dir, 5)

        if not backward.is_empty():
            points.append(get_cell_world_position(cell))
            points.append(get_cell_world_position(backward[-1]))
            return points

    # 实在找不到巡逻路径，就原地站立
    points.append(get_cell_world_position(cell))
    return points


func _find_room_containing_cell(cell: Vector2i) -> Rect2i:
    for room in rooms:
        if _room_has_cell(room, cell):
            return room

    return Rect2i()


func _room_has_cell(room: Rect2i, cell: Vector2i) -> bool:
    return (
        cell.x >= room.position.x
        and cell.x < room.position.x + room.size.x
        and cell.y >= room.position.y
        and cell.y < room.position.y + room.size.y
    )


func _scan_walkable_cells(from_cell: Vector2i, dir: Vector2i, max_steps: int) -> Array[Vector2i]:
    var result: Array[Vector2i] = []

    var cell := from_cell + dir

    for i in max_steps:
        if is_cell_walkable(cell):
            result.append(cell)
            cell += dir
        else:
            break

    return result
```

# 8. 本课核心知识讲解

---

## 8.1 敌人为什么使用巡逻点？

本课敌人的行为非常简单：

```text
走到巡逻点 A
等待
走到巡逻点 B
等待
循环
```

这是敌人 AI 的最小原型。

后续可以扩展成：

```text
待机
巡逻
发现玩家
追击
攻击
受击
死亡
掉落
```

但无论多复杂，状态机通常都是从这种简单状态开始的。

---

## 8.2 敌人为什么不直接用物理碰撞寻路？

本课敌人移动仍然依赖：

```gdscript
dungeon.is_world_position_walkable()
```

原因是：

- 简单。
- 和地图数据强绑定。
- 容易调试。
- 不需要额外烘焙导航。
- 适合小规模 2D 地牢。

后续你可以升级成：

- `NavigationAgent2D`
- `AStarGrid2D` 路径
- 房间图巡逻
- 路径点图
- 视线检测
- 动态避障

---

## 8.3 敌人巡逻路径是怎么生成的？

当前策略：

```text
如果敌人在房间里：
    巡逻点 = 敌人出生点 + 房间中心
否则：
    沿一个可走方向扫描最多 5 格
    巡逻点 = 敌人出生点 + 扫描终点
如果都失败：
    敌人原地站立
```

这个策略很适合房间式地牢。

它会让敌人看起来像是在房间里巡视。

---

## 8.4 玩家生命值的关键点

玩家脚本里现在有三个关键状态：

```text
health
invincible
knockback
```

含义：

```text
health       当前生命
invincible   是否无敌
knockback    击退速度
```

受伤流程：

```text
take_damage()
  ├─ 如果无敌，忽略
  ├─ 扣血
  ├─ 如果死亡，调用 die()
  ├─ 否则击退
  └─ 进入无敌时间
```

这是动作游戏里非常常见的最小受伤系统。

---

## 8.5 为什么受伤后要有无敌时间？

如果没有无敌时间：

```text
玩家碰到敌人
一帧内连续触发多次伤害
玩家瞬间死亡
```

所以我们需要：

```text
受伤后短时间内不再受伤
```

这就是常说的：

```text
无敌帧
i-frame
invincibility frames
```

本课用：

```gdscript
invincible
modulate.a
await Timer
```

实现了一个简单版本。

# 9. 常见问题排查

---

## 问题 1：敌人没有出现

检查：

1. `Main` 的 `Enemy Scene` 是否已经拖入 `enemy.tscn`。
2. `min_monsters` 和 `max_monsters` 是否大于 0。
3. `enemy.tscn` 根节点是否是 `CharacterBody2D`。
4. 控制台是否报错。

如果没有设置 `Enemy Scene`，本课会退回第 4 课的紫色方块。

---

## 问题 2：敌人不动

可能原因：

1. `patrol_points` 只有一个点。
2. `dungeon` 没有传入。
3. 敌人前方被地图数据判断为不可走。
4. `collision_radius` 太大，导致敌人一直认为前方被墙挡住。

可以尝试：

```text
Enemy collision_radius = 3
Enemy speed = 80
```

---

## 问题 3：敌人穿墙

检查：

1. `Enemy` 的 `collision_radius` 是否太小。
2. `Main` 的 `cell_size` 是否和瓦片大小一致。
3. 敌人速度是否过快。
4. 地图格子是否真的封闭。

本课敌人不是靠物理墙阻挡，而是靠地图查询。

---

## 问题 4：玩家碰到敌人不掉血

检查：

1. `Enemy` 场景里是否有 `Hitbox`。
2. `Hitbox` 是否是 `Area2D`。
3. `Hitbox` 是否有 `CollisionShape2D`。
4. 玩家是否在 `player` 组。
5. 玩家脚本是否有 `take_damage()`。
6. `Hitbox` 的 collision mask 是否包含玩家 layer。

推荐设置：

```text
Player Collision Layer = 1
Player Collision Mask = 1

Enemy Root Collision Layer = 2
Enemy Root Collision Mask = 0

Enemy Hitbox Collision Layer = 0
Enemy Hitbox Collision Mask = 1
```

---

## 问题 5：敌人挡住玩家

如果你发现玩家被敌人物理挡住，检查：

```text
Enemy Root Collision Mask = 0
```

也可以把：

```text
Enemy Root Collision Layer = 2
```

这样玩家默认不会和敌人发生物理碰撞。

伤害仍然由 `Hitbox` 检测。

---

## 问题 6：玩家一直掉血

通常是没有无敌时间，或者无敌时间没有生效。

检查：

```gdscript
if invincible:
    return
```

是否在 `take_damage()` 最前面。

以及：

```gdscript
_start_invincibility(invincibility_time)
```

是否被调用。
