# 第 4 课：钥匙、宝箱、怪物出生点、出口锁

本课目标：

1. 在地牢中生成钥匙。
2. 没有钥匙时，出口不可用。

## 6.1 添加 POI 主函数

```gdscript
func _pick_poi_cells() -> void:
    used_cells.clear()

    key_cell = INVALID_CELL
    treasure_cells.clear()
    monster_cells.clear()

    # 入口和出口不能被内容点占用
    _mark_cell_used(entrance_cell)
    _mark_cell_used(exit_cell)

    # 先选钥匙
    key_cell = _pick_key_cell()
    if key_cell != INVALID_CELL:
        _mark_cell_used(key_cell)

    # 再选宝箱
    var min_t := mini(min_treasures, max_treasures)
    var max_t := maxi(min_treasures, max_treasures)
    var treasure_amount := rng.randi_range(min_t, max_t)

    for i in treasure_amount:
        var cell := _pick_random_available_cell_in_rooms()
        if cell == INVALID_CELL:
            break

        treasure_cells.append(cell)
        _mark_cell_used(cell)

    # 最后选怪物
    var min_m := mini(min_monsters, max_monsters)
    var max_m := maxi(min_monsters, max_monsters)
    var monster_amount := rng.randi_range(min_m, max_m)

    for i in monster_amount:
        var cell := _pick_random_available_cell_in_rooms()
        if cell == INVALID_CELL:
            break

        monster_cells.append(cell)
        _mark_cell_used(cell)
```

## 6.2 添加格子占用与可用性判断

```gdscript
func _mark_cell_used(cell: Vector2i) -> void:
    if cell == INVALID_CELL:
        return

    used_cells[cell] = true


func _is_cell_available(cell: Vector2i) -> bool:
    if cell == INVALID_CELL:
        return false

    if cell.x < 0 or cell.x >= map_width:
        return false
    if cell.y < 0 or cell.y >= map_height:
        return false

    if grid[cell.y][cell.x] != CELL_FLOOR:
        return false

    if used_cells.has(cell):
        return false

    return true
```

## 6.3 添加钥匙位置选择

钥匙的策略：

```text
优先放在离入口较远的房间。
不要放在入口。
不要放在出口。
不要和宝箱、怪物重叠。
```

添加代码：

```gdscript
func _pick_key_cell() -> Vector2i:
    if rooms.is_empty():
        return INVALID_CELL

    var candidate_rooms: Array = []

    # 优先排除入口房和出口房
    for room in rooms:
        var center := _room_center(room)

        if center == entrance_cell:
            continue
        if center == exit_cell:
            continue

        candidate_rooms.append(room)

    if candidate_rooms.is_empty():
        candidate_rooms = rooms

    # 在候选房间里选择离入口最远的房间
    var best_room: Rect2i = candidate_rooms[0]
    var best_distance := -1

    for room in candidate_rooms:
        var center := _room_center(room)
        var distance := entrance_cell.distance_squared_to(center)

        if distance > best_distance:
            best_distance = distance
            best_room = room

    var cell := _pick_random_available_cell_in_room(best_room)

    if cell != INVALID_CELL:
        return cell

    # 如果这个房间没有合适位置，就全局找一个
    return _pick_random_available_cell_in_grid()
```

## 6.4 添加随机可用格子选择

下面这些函数会帮助我们在房间或整个地图里找可用格子。

```gdscript
func _pick_random_available_cell_in_room(room: Rect2i) -> Vector2i:
    var candidates: Array[Vector2i] = []

    for y in range(room.position.y, room.position.y + room.size.y):
        for x in range(room.position.x, room.position.x + room.size.x):
            var cell := Vector2i(x, y)

            if _is_cell_available(cell):
                candidates.append(cell)

    if candidates.is_empty():
        return INVALID_CELL

    return candidates[rng.randi_range(0, candidates.size() - 1)]


func _pick_random_available_cell_in_rooms() -> Vector2i:
    var candidates: Array[Vector2i] = []

    for room in rooms:
        for y in range(room.position.y, room.position.y + room.size.y):
            for x in range(room.position.x, room.position.x + room.size.x):
                var cell := Vector2i(x, y)

                if _is_cell_available(cell):
                    candidates.append(cell)

    if candidates.is_empty():
        return _pick_random_available_cell_in_grid()

    return candidates[rng.randi_range(0, candidates.size() - 1)]


func _pick_random_available_cell_in_grid() -> Vector2i:
    var candidates: Array[Vector2i] = []

    for y in map_height:
        for x in map_width:
            var cell := Vector2i(x, y)

            if _is_cell_available(cell):
                candidates.append(cell)

    if candidates.is_empty():
        return INVALID_CELL

    return candidates[rng.randi_range(0, candidates.size() - 1)]
```

```gdscript
func _create_pickup_area(
    cell: Vector2i,
    pickup_type: String,
    color: Color,
    radius_multiplier: float
) -> Area2D:
    var area := _create_poi_area(cell, color, radius_multiplier)

    area.add_to_group(pickup_type)

    if pickup_type == "key":
        area.body_entered.connect(_on_key_body_entered.bind(area))
    elif pickup_type == "treasure":
        area.body_entered.connect(_on_treasure_body_entered.bind(area))

    return area
```

## 7.4 创建通用 POI Area2D

```gdscript
func _create_poi_area(
    cell: Vector2i,
    color: Color,
    radius_multiplier: float
) -> Area2D:
    var area := Area2D.new()
    area.position = _cell_to_center_local(cell)
    area.monitoring = true

    var collision := CollisionShape2D.new()
    var circle := CircleShape2D.new()
    circle.radius = cell_size * radius_multiplier
    collision.shape = circle
    area.add_child(collision)

    var visual := Polygon2D.new()
    var size := cell_size * radius_multiplier

    visual.polygon = PackedVector2Array([
        Vector2(-size, -size),
        Vector2(size, -size),
        Vector2(size, size),
        Vector2(-size, size)
    ])

    visual.color = color
    area.add_child(visual)

    tile_layer.add_child(area)
    dynamic_entities.append(area)

    return area
```

# 11. 本课核心知识讲解

---

## 11.1 地图生成开始变成内容生成

前三课我们主要生成：

```text
地形
房间
走廊
入口
出口
连通性
```

这一课开始生成：

```text
游戏内容
```

也就是：

```text
钥匙
宝箱
怪物
任务目标
```

这是 RPG 程序化地图真正重要的部分。

地形只是舞台，内容才是玩法。

---

## 11.2 used_cells 的作用

我们使用：

```gdscript
var used_cells: Dictionary = {}
```

记录已经占用的格子。

当前占用规则：

```text
入口已占用
出口已占用
钥匙已占用
每个宝箱已占用
每个怪物已占用
```

这样可以避免：

- 钥匙和宝箱重叠。
- 怪物出生在出口。
- 宝箱出生在入口。
- 多个内容点挤在同一格。

后续你可以继续扩展：

```text
商人不能离入口太近
Boss 房不能有小怪
任务物品不能放在死路
宝箱尽量放在支路
钥匙放在关键路径上
```

---

## 11.3 为什么钥匙优先放远房间？

当前钥匙策略是：

```text
排除入口房和出口房
在剩余房间中选择离入口最远的房间
```

这会让玩家不得不探索地牢。

如果钥匙就在入口旁边，地牢探索感会变弱。

后续可以升级成：

- 用 AStar 计算实际路径长度。
- 把钥匙放在“实际走起来最远”的房间。
- 把钥匙放在需要绕路的支路房间。
- 把钥匙放在怪物后面。
- 把钥匙放在宝箱后面。
- 把钥匙放在需要能力解锁的区域。

---

## 11.4 为什么怪物只是 Area2D？

本课的怪物不是完整敌人。

它是一个：

```text
危险区域
```

目的是先建立游戏规则：

```text
玩家碰到怪物会受到惩罚
```

后续我们会把它升级成真正的敌人：

```text
敌人场景
敌人状态机
巡逻路径
视野检测
追击玩家
攻击玩家
生命值
死亡掉落
```

# 12. 常见问题排查

---

## 问题 1：钥匙、宝箱、怪物没有出现

检查：

1. `_spawn_poi_nodes()` 是否在 `generate()` 里被调用。
2. `min_treasures` / `max_treasures` 是否大于 0。
3. `min_monsters` / `max_monsters` 是否大于 0。
4. 地牢是否生成了房间。
5. 控制台是否报错。

你也可以打开：

```text
Debug Print Path
```

然后看控制台是否打印：

```text
钥匙：
宝箱数：
怪物数：
```

---

## 问题 2：出口一直显示需要钥匙

检查：

1. 玩家是否真的碰到了黄色钥匙。
2. 控制台是否打印：

```text
拿到了钥匙！出口已解锁。
```

如果没有打印，说明钥匙没有被触发。

可能原因：

- 玩家没有进入钥匙区域。
- 钥匙区域碰撞半径太小。
- 玩家不在 `player` 组。
- `Area2D` 的 collision mask 和玩家 collision layer 不匹配。

---

## 问题 3：玩家一开始就碰到怪物

通常是因为怪物出生点离入口太近。

当前版本只是随机可用格子。

你可以尝试降低怪物数量：

```text
Min Monsters = 0
Max Monsters = 2
```

后续作业可以让怪物不要出生在入口附近。

---

## 问题 4：钥匙和宝箱重叠

正常情况下不会重叠，因为我们用了：

```gdscript
used_cells
```

如果你自己修改了代码，注意：

每次放置一个内容点后，都要调用：

```gdscript
_mark_cell_used(cell)
```

---

## 问题 5：重新生成后旧的钥匙/宝箱还在

检查：

```gdscript
_clear_dynamic_entities()
```

是否在 `generate()` 里被调用。

它应该在这些内容之前调用：

```gdscript
_pick_poi_cells()
_spawn_poi_nodes()
```

```text
验证入口到每个怪物是否可达。
```

使用：

```gdscript
astar_grid.get_id_path(entrance_cell, key_cell)
```

如果不可达，就重新选择钥匙位置。

```text
enemy.tscn
```

```text
CharacterBody2D
```

```gdscript
enemy_scene.instantiate()
```
